 10    def test_simple_nfa_conversion(self):
 11        """Test conversion of a simple NFA without epsilon transitions"""
 12        # NFA that accepts strings ending with 'ab'
 13        nfa = {
 14            'states': ['S0', 'S1', 'S2'],
 15            'alphabet': ['a', 'b'],
 16            'transitions': {
 17                'S0': {'a': ['S0', 'S1'], 'b': ['S0']},  # Non-deterministic on 'a'
 18                'S1': {'b': ['S2']},
 19                'S2': {}
 20            },
 21            'startingState': 'S0',
 22            'acceptingStates': ['S2']
 23        }
 24
 25        dfa = nfa_to_dfa(nfa)
 26
 27        # Verify DFA properties
 28        self.assertTrue(is_deterministic(dfa))
 29        self.assertEqual(dfa['alphabet'], ['a', 'b'])
 30        self.assertIn(dfa['startingState'], dfa['states'])
 31
 32        # Verify language equivalence on test strings
 33        test_strings = ['', 'a', 'b', 'ab', 'ba', 'aab', 'abb', 'abab', 'baba']
 34        for test_string in test_strings:
 35            nfa_result = simulate_nondeterministic_fsa(nfa, test_string)
 36            dfa_result = simulate_deterministic_fsa(dfa, test_string)
 37
 38            nfa_accepted = isinstance(nfa_result, list)
 39            dfa_accepted = isinstance(dfa_result, list)
 40
 41            self.assertEqual(nfa_accepted, dfa_accepted,
 42                             f"Disagreement on string '{test_string}'")

278    def test_conversion_result_structure(self):
279        """Test that converted DFA has correct structure"""
280        nfa = {
281            'states': ['S0', 'S1', 'S2'],
282            'alphabet': ['a', 'b'],
283            'transitions': {
284                'S0': {'a': ['S0', 'S1'], 'b': ['S0']},
285                'S1': {'b': ['S2']},
286                'S2': {}
287            },
288            'startingState': 'S0',
289            'acceptingStates': ['S2']
290        }
291
292        dfa = nfa_to_dfa(nfa)
293
294        # Check all required keys are present
295        required_keys = ['states', 'alphabet', 'transitions', 'startingState', 'acceptingStates']
296        for key in required_keys:
297            self.assertIn(key, dfa)
298
299        # Check types
300        self.assertIsInstance(dfa['states'], list)
301        self.assertIsInstance(dfa['alphabet'], list)
302        self.assertIsInstance(dfa['transitions'], dict)
303        self.assertIsInstance(dfa['startingState'], str)
304        self.assertIsInstance(dfa['acceptingStates'], list)
305
306        # Check that start state is in states list
307        self.assertIn(dfa['startingState'], dfa['states'])
308
309        # Check that all accepting states are in states list
310        for state in dfa['acceptingStates']:
311            self.assertIn(state, dfa['states'])
312
313        # Check that states are sorted
314        self.assertEqual(dfa['states'], sorted(dfa['states']))
315
316        # Check that accepting states are sorted
317        self.assertEqual(dfa['acceptingStates'], sorted(dfa['acceptingStates']))
