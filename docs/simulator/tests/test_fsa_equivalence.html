 15    def test_normalise_automaton_nfa_to_dfa(self):
 16        """Test normalising NFA to minimal DFA"""
 17        # Simple NFA that accepts strings ending with 'ab'
 18        nfa = {
 19            'states': ['S0', 'S1', 'S2'],
 20            'alphabet': ['a', 'b'],
 21            'transitions': {
 22                'S0': {'a': ['S0', 'S1'], 'b': ['S0']},
 23                'S1': {'b': ['S2']},
 24                'S2': {}
 25            },
 26            'startingState': 'S0',
 27            'acceptingStates': ['S2']
 28        }
 29
 30        result = normalise_automaton(nfa)
 31
 32        # Should return a DFA
 33        self.assertTrue(is_deterministic(result))
 34        self.assertIsInstance(result, dict)
 35        self.assertIn('states', result)
 36        self.assertIn('alphabet', result)
 37        self.assertIn('transitions', result)
 38        self.assertIn('startingState', result)
 39        self.assertIn('acceptingStates', result)
 40
 41        # Test language equivalence
 42        test_strings = ['', 'a', 'b', 'ab', 'ba', 'aab', 'abb', 'abab']
 43        for test_string in test_strings:
 44            original_result = simulate_nondeterministic_fsa(nfa, test_string)
 45            normalised_result = simulate_deterministic_fsa(result, test_string)
 46
 47            original_accepted = isinstance(original_result, list)
 48            normalised_accepted = isinstance(normalised_result, list)
 49
 50            self.assertEqual(original_accepted, normalised_accepted,
 51                             f"Disagreement on string '{test_string}'")
