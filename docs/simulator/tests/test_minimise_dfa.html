 10    def test_simple_dfa_minimisation(self):
 11        """Test minimisation of a simple DFA with equivalent states"""
 12        # DFA that accepts strings ending with 'a'
 13        # States S0, S1 and S2 are all equivalent (all non-accepting, same transition behavior)
 14        dfa = {
 15            'states': ['S0', 'S1', 'S2', 'S3'],
 16            'alphabet': ['a', 'b'],
 17            'transitions': {
 18                'S0': {'a': ['S3'], 'b': ['S1']},  # Start state
 19                'S1': {'a': ['S3'], 'b': ['S2']},  # Non-accepting
 20                'S2': {'a': ['S3'], 'b': ['S1']},  # Non-accepting (equivalent to S0 and S1)
 21                'S3': {'a': ['S3'], 'b': ['S1']}  # Accepting
 22            },
 23            'startingState': 'S0',
 24            'acceptingStates': ['S3']
 25        }
 26
 27        minimised = minimise_dfa(dfa)
 28

 31
 32        self.assertEqual(len(minimised['states']), 2)
 33        self.assertTrue(is_deterministic(minimised))
 34        self.assertEqual(minimised['alphabet'], ['a', 'b'])
 35        self.assertEqual(len(minimised['acceptingStates']), 1)
 36
 37        # Test that both DFAs accept the same language
 38        test_strings = ['', 'a', 'b', 'ab', 'ba', 'aa', 'bb', 'aba', 'bab']
 39        for test_string in test_strings:

 42
 43            if isinstance(original_result, list):
 44                original_accepted = True
 45            else:
